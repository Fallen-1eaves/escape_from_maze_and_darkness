int sceneCount = sizeof(scenes) / sizeof(scenes[0]);
int main() {
#ifdef _WIN32
    SetConsoleOutputCP(CP_UTF8);
#endif
    int currentScene = 0;
    int choice;
    printf("输入选项前的数字\n");

    while (currentScene >= 0 && currentScene < sceneCount) {
        struct scene *s = &scenes[currentScene];
        printf("---------------------------------------\n");
        printf("%s\n\n", s->de);
        printf("---------------------------------------\n");
        // 显示当前状态（方便玩家参考）
        printf("[fear：%d  happy：%d]\n", st.fear, st.happy);
        int available[MAX_OPS];
        int availCount = 0;
        for (int i = 0; i < s->num1; i++) {
            OP *op = &s->ops[i];
            if ((op->Needfear == -1 || st.fear >= op->Needfear) &&
                (op->Needhappy == -1 || st.happy >= op->Needhappy)) {
                available[availCount] = i;
                printf("%d. %s\n", availCount + 1, op->text);
                availCount++;
            }
        }

        if (availCount == 0) {
            printf("游戏结束。\n");
            break;
        }
        printf("\n请选择 (1-%d): ", availCount);
        char input[10];
        fgets(input, sizeof(input), stdin);
        choice = atoi(input);
        if (choice < 1 || choice > availCount) {
            printf("error\n\n");
            continue;
		}
		int realIdx = available[choice - 1];
        OP *selected = &s->ops[realIdx];

        if (selected->action != NULL) {
            selected->action(&st);
        }

        currentScene = selected->next;
        printf("\n");
    }

    printf("游戏结束\n");
    return 0;
}
